package main

import "fmt"

type Interface interface {
	method()
}

func fn(v Interface) {
	v.method()
}

func main() {
	t := T{}
	// fmt.Printf("%p\n", &t)
	fn(t)


	fmt.Println("Payload:", t.Payload)
}

type T struct {
	Payload string
}

func (t T) method() {
	// fmt.Printf("%p\n", &t)
	t.Payload = "Value!"
	fmt.Println("Method called")
}
